1. EXTENSION METHODS


// NAMA  : MARTA SAPUTRA
// NIM   : 251410020
// KELAS : SI2A

// Extension Method pada String
extension StringExt on String {

  // Mengubah huruf pertama menjadi kapital
  String capitalize() {
    return this[0].toUpperCase() + substring(1);
  }

  // Mengecek apakah panjang teks genap
  bool get isEvenLength {
    return length % 2 == 0;
  }
}

// Extension Method pada int
extension IntExt on int {

  // Mengecek bilangan genap
  bool get isEven => this % 2 == 0;

  // Menghitung kuadrat
  int square() => this * this;
}

void main() {

  print("===== TUGAS EXTENSION METHODS =====");
  print("Nama  : Marta Saputra");
  print("NIM   : 251410020");
  print("Kelas : SI2A");
  print("==================================");

  // String Extension
  String nama = "marta saputra";

  print("\nExtension String:");
  print("Asli       : $nama");
  print("Capitalize : ${nama.capitalize()}");
  print("Panjang Genap? ${nama.isEvenLength}");

  // Int Extension
  int angka = 10;

  print("\nExtension Integer:");
  print("Angka            : $angka");
  print("Bilangan Genap?  : ${angka.isEven}");
  print("Kuadrat          : ${angka.square()}");
}
