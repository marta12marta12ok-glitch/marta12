1. EXTENSION METHODS

Polimorfisme pada class hewan adalah konsep dalam pemrograman di mana satu metode yang sama dapat digunakan oleh berbagai jenis hewan tetapi menghasilkan perilaku yang berbeda sesuai dengan jenis hewannya.


// NAMA : MARTA SAPUTRA
// NIM  : 251410020
// KELAS: SI2A

// Extension Method pada String
extension StringExt on String {

  // Mengubah huruf pertama menjadi kapital
  String capitalize() {
    return this[0].toUpperCase() + substring(1);
  }

  // Mengecek apakah panjang teks genap
  bool get isEvenLength {
    return length % 2 == 0;
  }
}

// Extension Method pada int
extension IntExt on int {

  // Mengecek bilangan genap
  bool get isEven => this % 2 == 0;

  // Menghitung kuadrat
  int square() => this * this;
}

void main() {

  print("===== TUGAS EXTENSION METHODS =====");
  print("Nama  : Marta Saputra");
  print("NIM   : 251410020");
  print("Kelas : SI2A");
  print("==================================");

  // String Extension
  String nama = "marta saputra";

  print("\nExtension String:");
  print("Asli       : $nama");
  print("Capitalize : ${nama.capitalize()}");
  print("Panjang Genap? ${nama.isEvenLength}");

  // Int Extension
  int angka = 10;

  print("\nExtension Integer:");
  print("Angka       : $angka");
  print("Bilangan Genap? ${angka.isEven}");
  print("Kuadrat     : ${angka.square()}");
}

2. PENGANTAR SINTAKS FACTORY CONSTRUCTOR

Class Bentuk pada polimorfisme adalah konsep di mana satu metode yang sama pada berbagai bentuk (seperti lingkaran, persegi, atau segitiga) dapat menghasilkan perilaku atau perhitungan yang berbeda sesuai dengan jenis bentuknya.


// NAMA : MARTA SAPUTRA
// NIM  : 251410020
// KLS  : SI2A

class Mahasiswa {
  String nama;
  String nim;

  // Menyimpan instance yang sudah dibuat
  static final Map _cache = {};

  // Constructor private
  Mahasiswa._internal(this.nama, this.nim);

  // Factory Constructor
  factory Mahasiswa(String nama, String nim) {
    if (_cache.containsKey(nim)) {
      print("Instance dengan NIM $nim sudah ada.");
      return _cache[nim]!;
    } else {
      print("Membuat instance baru untuk NIM $nim.");
      final mahasiswa = Mahasiswa._internal(nama, nim);
      _cache[nim] = mahasiswa;
      return mahasiswa;
    }
  }

  void tampilkanInfo() {
    print("Nama : $nama");
    print("NIM  : $nim");
  }
}

void main() {
  var mhs1 = Mahasiswa("MARTA SAPUTRA", "251410020");
  var mhs2 = Mahasiswa("MARTA SAPUTRA", "251410020");

  print("\nData Mahasiswa:");
  mhs1.tampilkanInfo();

  print("\nApakah kedua objek sama?");
  print(identical(mhs1, mhs2)); // true
}

4. SINGLETON PATTERN

Abstract class Kendaraan adalah class dasar yang tidak dapat dibuat objeknya secara langsung dan digunakan sebagai dasar bagi class turunan seperti mobil atau motor.


// NAMA : MARTA SAPUTRA
// NIM  : 251410020
// KELAS: SI2A

class DatabaseConnection {
  // Private Constructor
  DatabaseConnection._internal() {
    print("Creating database connection...");
  }

  // Static instance (hanya satu)
  static final DatabaseConnection _instance =
      DatabaseConnection._internal();

  // Factory Constructor
  factory DatabaseConnection() {
    return _instance;
  }

  void connect() {
    print("Connected to database");
  }
}

void main() {
  // Membuat dua objek
  var db1 = DatabaseConnection();
  var db2 = DatabaseConnection();

  // Mengecek apakah keduanya objek yang sama
  print("Apakah db1 dan db2 sama?");
  print(identical(db1, db2));

  // Menggunakan method connect
  db1.connect();
  db2.connect();
}

5. OBJECT POOL/CACHING

Class AlatMusik dalam abstraksi adalah konsep yang menyembunyikan detail implementasi dan hanya menampilkan fungsi penting seperti cara memainkan alat musik.


// NAMA : MARTA SAPUTRA
// NIM  : 251410020
// KELAS: SI2A

class ConnectionPool {
  final String host;
  final int port;

  // Cache berdasarkan parameter
  static final Map _cache = {};

  // Private Constructor
  ConnectionPool._(this.host, this.port);

  // Factory Constructor
  factory ConnectionPool(String host, int port) {
    var key = '$host:$port';

    // Return dari cache jika sudah ada
    if (_cache.containsKey(key)) {
      print('Returning cached connection for $key');
      return _cache[key]!;
    }

    // Membuat koneksi baru
    print('Creating new connection for $key');
    var connection = ConnectionPool._(host, port);
    _cache[key] = connection;
    return connection;
  }

  void query(String sql) {
    print('Executing "$sql" on $host:$port');
  }
}

void main() {
  print('--- Membuat conn1 (localhost:5432) ---');
  var conn1 = ConnectionPool('localhost', 5432);
  conn1.query('SELECT * FROM users');

  print('');

  print('--- Membuat conn2 (localhost:5432) ---');
  var conn2 = ConnectionPool('localhost', 5432);
  conn2.query('INSERT INTO products VALUES ("Laptop")');

  print('');

  print('--- Membuat conn3 (db.example.com:3306) ---');
  var conn3 = ConnectionPool('db.example.com', 3306);
  conn3.query('UPDATE orders SET status="shipped" WHERE id=101');

  print('');

  print('Apakah conn1 dan conn2 sama? ${identical(conn1, conn2)}');
  print('Apakah conn1 dan conn3 sama? ${identical(conn1, conn3)}');
}

6. MENGEMBALIKAN SUBCLASS

Konsep interface dalam Dart digunakan untuk mendefinisikan kontrak yang harus diimplementasikan oleh class lain.


// NAMA : MARTA SAPUTRA
// NIM  : 251410020
// KELAS: SI2A

abstract class Shape {
  void draw();

  // Factory Constructor
  factory Shape(String type) {
    switch (type.toLowerCase()) {
      case 'circle':
        return Circle();
      case 'square':
        return Square();
      case 'triangle':
        return Triangle();
      case 'rectangle':
        return Rectangle();
      default:
        throw ArgumentError('Unknown shape: $type');
    }
  }
}

class Circle implements Shape {
  @override
  void draw() {
    print('Drawing Circle');
  }
}

class Square implements Shape {
  @override
  void draw() {
    print('Drawing Square');
  }
}

class Triangle implements Shape {
  @override
  void draw() {
    print('Drawing Triangle');
  }
}

class Rectangle implements Shape {
  @override
  void draw() {
    print('Drawing Rectangle');
  }
}

void main() {
  Shape circle = Shape('circle');
  Shape square = Shape('square');
  Shape triangle = Shape('triangle');
  Shape rectangle = Shape('rectangle');

  circle.draw();
  square.draw();
  triangle.draw();
  rectangle.draw();
}

14. Utility Math (Bonus 5)

Class yang menyediakan fungsi-fungsi bantuan untuk perhitungan matematika.


// ==========================================
// Nama  : Marta Saputra
// NIM   : 251410020
// Kelas : SI2A
// ==========================================
/// Bonus 5

import 'dart:math';

class MathUtils {

  static Random rand = Random(1);

  static int faktorial(int n) {
    if(n <= 1) return 1;
    return n * faktorial(n-1);
  }

  static bool cekPrima(int n) {
    if(n <= 1) return false;
    for(int i = 2; i <= sqrt(n); i++) {
      if(n % i == 0) return false;
    }
    return true;
  }

  static double pembulatan(double angka, int desimal) {
    double scale = pow(10, desimal).toDouble();
    return (angka * scale).round() / scale;
  }

  static int randomInt(int min, int max) {
    return min + rand.nextInt(max - min + 1);
  }

  static double rataRata(List list) {
    int total = list.reduce((a, b) => a + b);
    return total / list.length;
  }

  static double median(List list) {
    List sorted = List.from(list)..sort();
    int n = sorted.length;

    if(n % 2 == 0) {
      return (sorted[n ~/ 2 - 1] + sorted[n ~/ 2]) / 2;
    } else {
      return sorted[n ~/ 2].toDouble();
    }
  }
}

void main() {
  print("=== Utility Math ===");
  print("Nama  : Marta Saputra");
  print("NIM   : 251410020");
  print("Kelas : SI2A");
  print("--------------------------------");

  print("Faktorial 5 = ${MathUtils.faktorial(5)}");
  print("Prima 7 = ${MathUtils.cekPrima(7)}");
  print("Pembulatan = ${MathUtils.pembulatan(3.14159, 2)}");
  print("Contoh angka = ${MathUtils.randomInt(1, 10)}");
  print("Rata-rata = ${MathUtils.rataRata([10, 20, 30, 40, 50])}");
  print("Median = ${MathUtils.median([10, 20, 30, 40, 50])}");
}
/*
Hasil Output:
Faktorial 5 = 120
Prima 7 = true
Pembulatan = 3.14
contoh angka = 3
Rata-rata = 30
Median = 30
*/

15. Kesimpulan

Dari latihan-latihan di atas, dapat disimpulkan bahwa:

  • Abstraksi menyembunyikan detail implementasi dan hanya menampilkan fungsi penting kepada pengguna.
  • Polimorfisme memungkinkan kita menggunakan satu nama method untuk melakukan berbagai tindakan yang berbeda tergantung objeknya.
  • Inheritance (Pewarisan) memungkinkan pembuatan class baru berdasarkan class yang sudah ada, sehingga kode lebih efisien dan terstruktur.
  • Interface berfungsi sebagai kontrak yang memaksa class turunan untuk mengimplementasikan semua method yang didefinisikan di dalamnya.
  • Dengan menggabungkan konsep-konsep OOP (Object-Oriented Programming) ini, program menjadi lebih mudah dikembangkan, dipelihara, dan dimodifikasi di kemudian hari.
