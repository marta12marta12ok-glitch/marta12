1. EXTENSION METHODS

Fitur yang memungkinkan kita menambahkan fungsi atau method baru ke suatu kelas tanpa mengubah kode asli kelas tersebut.


// NAMA : MARTA SAPUTRA

// KELAS: SI2A

// Extension Method pada String
extension StringExt on String {

  // Mengubah huruf pertama menjadi kapital
  String capitalize() {
    return this[0].toUpperCase() + substring(1);
  }

  // Mengecek apakah panjang teks genap
  bool get isEvenLength {
    return length % 2 == 0;
  }
}

// Extension Method pada int
extension IntExt on int {

  // Mengecek bilangan genap
  bool get isEven => this % 2 == 0;

  // Menghitung kuadrat
  int square() => this * this;
}

2. PENGANTAR SINTAKS FACTORY CONSTRUCTOR

Factory constructor adalah konstruktor khusus yang digunakan untuk mengontrol proses pembuatan objek, sehingga objek yang dibuat bisa berasal dari cache, singleton, atau subclass tertentu.


// NAMA : MARTA SAPUTRA
// NIM  : 251410020
// KLS  : SI2A

class Mahasiswa {
  String nama;
  String nim;

  // Menyimpan instance yang sudah dibuat
  static final Map _cache = {};

  // Constructor private
  Mahasiswa._internal(this.nama, this.nim);

  // Factory Constructor
  factory Mahasiswa(String nama, String nim) {
    if (_cache.containsKey(nim)) {
      print("Instance dengan NIM $nim sudah ada.");
      return _cache[nim]!;
    } else {
      print("Membuat instance baru untuk NIM $nim.");
      final mahasiswa = Mahasiswa._internal(nama, nim);
      _cache[nim] = mahasiswa;
      return mahasiswa;
    }
  }

  void tampilkanInfo() {
    print("Nama : $nama");
    print("NIM  : $nim");
  }
}

void main() {
  var mhs1 = Mahasiswa("MARTA SAPUTRA", "251410020");
  var mhs2 = Mahasiswa("MARTA SAPUTRA", "251410020");

  print("\nData Mahasiswa:");
  mhs1.tampilkanInfo();

  print("\nApakah kedua objek sama?");
  print(identical(mhs1, mhs2)); // true
}

4. SINGLETON PATTERN

Pola desain yang memastikan hanya ada satu instance objek dari suatu kelas selama aplikasi berjalan dan menyediakan akses global ke instance tersebut.


// NAMA : MARTA SAPUTRA
// NIM  : 251410020
// KELAS: SI2A

class DatabaseConnection {
  // Private Constructor
  DatabaseConnection._internal() {
    print("Creating database connection...");
  }

  // Static instance (hanya satu)
  static final DatabaseConnection _instance =
      DatabaseConnection._internal();

  // Factory Constructor
  factory DatabaseConnection() {
    return _instance;
  }

  void connect() {
    print("Connected to database");
  }
}

void main() {
  // Membuat dua objek
  var db1 = DatabaseConnection();
  var db2 = DatabaseConnection();

  // Mengecek apakah keduanya objek yang sama
  print("Apakah db1 dan db2 sama?");
  print(identical(db1, db2));

  // Menggunakan method connect
  db1.connect();
  db2.connect();
}

5. OBJECT POOL/CACHING

Teknik penyimpanan dan penggunaan kembali objek yang sudah dibuat sebelumnya agar tidak perlu membuat objek baru berulang kali, sehingga meningkatkan efisiensi penggunaan memori dan kinerja.


// NAMA : MARTA SAPUTRA
// NIM  : 251410020
// KELAS: SI2A

class ConnectionPool {
  final String host;
  final int port;

  // Cache berdasarkan parameter
  static final Map _cache = {};

  // Private Constructor
  ConnectionPool._(this.host, this.port);

  // Factory Constructor
  factory ConnectionPool(String host, int port) {
    var key = '$host:$port';

    // Return dari cache jika sudah ada
    if (_cache.containsKey(key)) {
      print('Returning cached connection for $key');
      return _cache[key]!;
    }

    // Membuat koneksi baru
    print('Creating new connection for $key');
    var connection = ConnectionPool._(host, port);
    _cache[key] = connection;
    return connection;
  }

  void query(String sql) {
    print('Executing "$sql" on $host:$port');
  }
}

void main() {
  print('--- Membuat conn1 (localhost:5432) ---');
  var conn1 = ConnectionPool('localhost', 5432);
  conn1.query('SELECT * FROM users');

  print('');

  print('--- Membuat conn2 (localhost:5432) ---');
  var conn2 = ConnectionPool('localhost', 5432);
  conn2.query('INSERT INTO products VALUES ("Laptop")');

  print('');

  print('--- Membuat conn3 (db.example.com:3306) ---');
  var conn3 = ConnectionPool('db.example.com', 3306);
  conn3.query('UPDATE orders SET status="shipped" WHERE id=101');

  print('');

  print('Apakah conn1 dan conn2 sama? ${identical(conn1, conn2)}');
  print('Apakah conn1 dan conn3 sama? ${identical(conn1, conn3)}');
}

6. MENGEMBALIKAN SUBCLASS

Factory constructor dapat mengembalikan objek dari kelas turunan (subclass) yang berbeda sesuai kondisi tertentu, meskipun tipe yang dikembalikan tetap kelas induknya.


// NAMA : MARTA SAPUTRA

abstract class Shape {

class Square implements Shape {

class Rectangle implements Shape {

7. FACTORY UNTUK CONGFIGURATION

Factory digunakan untuk membuat objek berdasarkan konfigurasi tertentu, seperti data dari file, database, atau pengaturan aplikasi, sehingga proses pembuatan objek menjadi lebih fleksibel.


// NAMA : MARTA SAPUTRA
// NIM  : 251410020
// KELAS: SI2A

class AppConfig {
  final String environment;
  final String apiUrl;
  final bool debugMode;

  // Constructor private untuk Development
  AppConfig._development()
      : environment = 'Development',
        apiUrl = 'http://localhost:3000',
        debugMode = true;

  // Constructor private untuk Production
  AppConfig._production()
      : environment = 'Production',
        apiUrl = 'https://api.example.com',
        debugMode = false;

  // Constructor private untuk Staging
  AppConfig._staging()
      : environment = 'Staging',
        apiUrl = 'https://staging.example.com',
        debugMode = true;

  // Factory Constructor
  factory AppConfig.fromEnvironment(String env) {
    switch (env.toLowerCase()) {
      case 'dev':
      case 'development':
        return AppConfig._development();

      case 'prod':
      case 'production':
        return AppConfig._production();

      case 'stage':
      case 'staging':
        return AppConfig._staging();

      default:
        throw ArgumentError('Environment tidak dikenal: $env');
    }
  }

  void printConfig() {
    print('Environment : $environment');
    print('API URL     : $apiUrl');
    print('Debug Mode  : $debugMode');
    print('-------------------------');
  }
}

void main() {
  var devConfig = AppConfig.fromEnvironment('dev');
  var prodConfig = AppConfig.fromEnvironment('prod');
  var stageConfig = AppConfig.fromEnvironment('stage');

  devConfig.printConfig();
  prodConfig.printConfig();
  stageConfig.printConfig();
}

9. FACTORY DENGAN GENERIC

Factory dapat menggunakan tipe generic () sehingga mampu membuat objek untuk berbagai tipe data tanpa perlu menulis kode yang berulang.


// NAMA : MARTA SAPUTRA
// NIM  : 251410020
// KLS  : SI2A

abstract class Response {
  final T? data;
  final String? error;

  Response({this.data, this.error});

  // Factory untuk sukses
  factory Response.success(T data) = SuccessResponse;

  // Factory untuk error
  factory Response.error(String error) = ErrorResponse;
}

// Class success
class SuccessResponse extends Response {
  SuccessResponse(T data) : super(data: data);
}

// Class error
class ErrorResponse extends Response {
  ErrorResponse(String error) : super(error: error);
}

void main() {

  // Data berhasil
  Response sukses =
      Response.success("Data Mahasiswa Berhasil Ditampilkan");

  // Data error
  Response gagal =
      Response.error("Data Mahasiswa Tidak Ditemukan");

  print("===== RESPONSE SUCCESS =====");
  print(sukses.data);

  print("");

  print("===== RESPONSE ERROR =====");
  print(gagal.error);
}

11. FACTORY VS STATIC METHOD

Factory constructor merupakan bagian dari konstruktor kelas dan dapat mengembalikan instance yang sudah ada atau subclass, sedangkan static method adalah fungsi statis yang tidak terkait langsung dengan proses konstruksi objek.


// NAMA : MARTA SAPUTRA
// NIM  : 251410020
// KLS  : SI2A

class Mahasiswa {
  String nama;
  String nim;
  String kelas;

  // Constructor biasa
  Mahasiswa(this.nama, this.nim, this.kelas);

  // Factory Constructor
  factory Mahasiswa.createMahasiswa(
      String nama, String nim, String kelas) {
    return Mahasiswa(nama, nim, kelas);
  }

  // Static Method
  static Mahasiswa createStatic(
      String nama, String nim, String kelas) {
    return Mahasiswa(nama, nim, kelas);
  }

  void tampilkanData() {
    print('Nama  : $nama');
    print('NIM   : $nim');
    print('Kelas : $kelas');
  }
}

void main() {
  // Menggunakan Factory Constructor
  var m1 = Mahasiswa.createMahasiswa(
      'MARTA SAPUTRA', '251410020', 'SI2A');

  // Menggunakan Static Method
  var m2 = Mahasiswa.createStatic(
      'MARTA SAPUTRA', '251410020', 'SI2A');

  print('=== Factory Constructor ===');
  m1.tampilkanData();

  print('\n=== Static Method ===');
  m2.tampilkanData();

  if (m1 != m2) {
    print('\nKedua objek merupakan instansi yang berbeda.');
  }
}

12. FACTORY UNTUK IMMUTABLE OBJECTS DENGAN CACHE

Factory constructor dapat menyimpan objek immutable yang sudah dibuat dalam cache, sehingga objek yang sama dapat digunakan kembali tanpa membuat instance baru.


// NAMA : MARTA SAPUTRA
// NIM  : 251410020
// KLS  : SI2A

class Mahasiswa {
  final String nama;
  final String nim;
  final String kelas;

  // Cache untuk menyimpan objek yang sudah dibuat
  static final Map _cache = {};

  // Private Constructor
  Mahasiswa._(this.nama, this.nim, this.kelas);

  // Factory Constructor dengan Cache
  factory Mahasiswa(String nama, String nim, String kelas) {
    String key = "$nama-$nim-$kelas";

    if (!_cache.containsKey(key)) {
      _cache[key] = Mahasiswa._(nama, nim, kelas);
    }

    return _cache[key]!;
  }

  @override
  String toString() {
    return "Nama: $nama, NIM: $nim, Kelas: $kelas";
  }
}

void main() {
  // Membuat objek pertama
  var mhs1 = Mahasiswa(
    "MARTA SAPUTRA",
    "251410020",
    "SI2A",
  );

  // Membuat objek kedua dengan data yang sama
  var mhs2 = Mahasiswa(
    "MARTA SAPUTRA",
    "251410020",
    "SI2A",
  );

  print("Data Mahasiswa:");
  print(mhs1);

  print("\nApakah mhs1 dan mhs2 objek yang sama?");
  print(identical(mhs1, mhs2)); // true karena menggunakan cache
}

14. RINGKASAN & BEST PRACTICES

Gunakan factory constructor ketika diperlukan kontrol terhadap pembuatan objek, caching, singleton, atau pemilihan subclass. Hindari penggunaan factory jika konstruktor biasa sudah cukup sederhana dan mudah dipahami.


// Nama  : Marta Saputra
// NIM   : 251410020
// Kelas : SI2A

class Mahasiswa {
  String nama;
  String nim;

  // Cache untuk menyimpan objek yang sudah dibuat
  static final Map _cache = {};

  // Generative Constructor
  Mahasiswa._internal(this.nama, this.nim);

  // Factory Constructor
  factory Mahasiswa(String nama, String nim) {
    // Validasi
    if (nama.isEmpty || nim.isEmpty) {
      throw ArgumentError("Nama dan NIM tidak boleh kosong!");
    }

    // Caching
    if (_cache.containsKey(nim)) {
      print("Mengambil data dari cache...");
      return _cache[nim]!;
    }

    print("Membuat objek baru...");
    final mahasiswaBaru = Mahasiswa._internal(nama, nim);

    _cache[nim] = mahasiswaBaru;

    return mahasiswaBaru;
  }

  void tampilkanData() {
    print("Nama : $nama");
    print("NIM  : $nim");
  }
}

void main() {
  var m1 = Mahasiswa("Marta Saputra", "251410020");
  var m2 = Mahasiswa("Marta Saputra", "251410020");

  print("\nData Mahasiswa:");
  m1.tampilkanData();

  print("\nApakah m1 dan m2 objek yang sama?");
  print(identical(m1, m2));
}

15. Kesimpulan

Dari latihan-latihan di atas, dapat disimpulkan bahwa:

    Factory Constructor pada Dart merupakan pola yang sangat berguna untuk mengontrol proses pembuatan objek. Dengan Factory Constructor, objek dapat dibuat berdasarkan kondisi tertentu, menggunakan cache agar objek yang sama tidak dibuat berulang kali, serta memilih subclass yang sesuai berdasarkan input pengguna. Selain itu, Factory Constructor juga dapat digunakan untuk validasi data sebelum objek dibuat sehingga program menjadi lebih efisien, terstruktur, dan mudah dikelola. Melalui latihan seperti ConnectionFactory, NotificationFactory, ShapeFactory, dan AnimalFactory, dapat dipahami bahwa Factory Constructor membantu menghemat penggunaan memori, meningkatkan performa aplikasi, serta mempermudah pengembangan kode yang fleksibel dan dapat digunakan kembali (reusable). Oleh karena itu, Factory Constructor menjadi salah satu fitur penting dalam pemrograman berorientasi objek menggunakan Dart.
